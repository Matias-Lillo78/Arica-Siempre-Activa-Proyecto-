        // 2. LÓGICA DE GRÁFICOS (Chart.js)

        function initCharts() {
            if(!chartEvolucion) {
                const ctxE = document.getElementById('evolucionChart').getContext('2d');
                chartEvolucion = new Chart(ctxE, { type: 'line', data: { labels: ['ENE', 'FEB', 'MAR', 'ABR', 'MAY', 'JUN'], datasets: [{ label: 'Inscripciones', data: [20, 35, 30, 45, 40, 60], borderColor: '#0D1F3C', backgroundColor: '#0D1F3C', borderWidth: 2, tension: 0.4 }, { label: 'Ingresos', data: [15, 25, 40, 35, 55, 50], borderColor: '#ea580c', backgroundColor: '#ea580c', borderWidth: 2, tension: 0.4 } ] }, options: { responsive: true, plugins: { legend: { display: false } }, scales: { y: { display: false }, x: { grid: { display: false } } } } });
            }
            if(!chartPagos) {
                const ctxP = document.getElementById('pagosChart').getContext('2d');
                chartPagos = new Chart(ctxP, { type: 'bar', data: { labels: ['Aprobados', 'Rechazados', 'Verif.'], datasets: [{ data: [262, 22, 9], backgroundColor: ['#0D1F3C', '#ef4444', '#f59e0b'], borderRadius: 4 }] }, options: { responsive: true, plugins: { legend: { display: false } }, scales: { y: { display: false }, x: { grid: { display: false } } } } });
            }

        // 3. NAVEGACIÓN SPA
        const botonesMenu = document.querySelectorAll(".btn-menu");
        const seccionesPanel = document.querySelectorAll(".seccion-panel");

        botonesMenu.forEach(boton => {
            boton.addEventListener("click", function(e) {
                e.preventDefault();
                botonesMenu.forEach(btn => btn.classList.remove("active"));
                this.classList.add("active");
                seccionesPanel.forEach(seccion => { seccion.classList.add("d-none"); seccion.classList.remove("fade-slide-in"); });

                seccionActiva.classList.remove("d-none");
                void seccionActiva.offsetWidth; 
                seccionActiva.classList.add("fade-slide-in");
                if(this.getAttribute("data-target") === "seccion-estadisticas") initCharts();
            });
        });

        // 4. MENÚ HAMBURGUESA
        document.querySelectorAll(".toggleSidebarBtn").forEach(btn => {
            btn.addEventListener("click", () => document.getElementById("sidebarPanel").classList.toggle("collapsed"));
        });
    });
